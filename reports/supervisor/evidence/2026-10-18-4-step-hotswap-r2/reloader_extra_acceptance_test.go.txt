package reloader

import (
    "database/sql"
    "os"
    "path/filepath"
    "testing"
    "time"
)

func TestR2SQLiteCommittedWALAndMultipleProjects(t *testing.T) {
    t.Setenv("USERPROFILE", t.TempDir())
    dir := t.TempDir()
    dbPath := filepath.Join(dir, "conversation_summaries.db")
    db, err := sql.Open("sqlite", dbPath)
    if err != nil { t.Fatal(err) }
    defer db.Close()
    db.SetMaxOpenConns(1)
    for _, query := range []string{
        "PRAGMA journal_mode=WAL",
        "PRAGMA wal_autocheckpoint=0",
        "CREATE TABLE conversation_summaries (conversation_id TEXT PRIMARY KEY, preview TEXT, workspace_uris TEXT, status TEXT)",
        "PRAGMA wal_checkpoint(TRUNCATE)",
    } {
        if _, err := db.Exec(query); err != nil { t.Fatal(err) }
    }
    records := []struct{ id, preview, workspace, status string }{
        {"aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa", "Command Working Directory: D:/Decoy\n", `["file:///C:/Project%20One"]`, "IDLE"},
        {"bbbbbbbb-bbbb-4bbb-8bbb-bbbbbbbbbbbb", "file:///C:/OtherDecoy", `["file:///D%3A/ProjectTwo"]`, "RUNNING"},
    }
    now := time.Now()
    for index, row := range records {
        if _, err := db.Exec("INSERT INTO conversation_summaries VALUES (?, ?, ?, ?)", row.id, row.preview, row.workspace, row.status); err != nil { t.Fatal(err) }
        supervisorFile(t, dir, row.id+".db", now.Add(-time.Duration(index+1)*time.Minute))
    }
    wal, err := os.Stat(dbPath+"-wal")
    if err != nil || wal.Size() == 0 { t.Fatalf("fixture must keep committed mappings in an uncheckpointed WAL: info=%v err=%v", wal, err) }
    threads, err := ScanActiveConversations2h(dir, dbPath, now)
    if err != nil || len(threads) != 2 { t.Fatalf("scan: %+v %v", threads, err) }
    if threads[0].ThreadID != records[0].id || threads[0].Project != "C:/Project One" || !threads[0].IsMainPane || threads[1].ThreadID != records[1].id || threads[1].Project != "D:/ProjectTwo" || threads[1].IsMainPane { t.Fatalf("WAL/row ownership/URI mapping: %+v", threads) }
    if got := resolveProjectFromSummaries(dbPath, "cccccccc-cccc-4ccc-8ccc-cccccccccccc"); got != "" { t.Fatalf("absent ID mapped to another record: %q", got) }
    t.Logf("committed WAL bytes=%d; IDLE plus RUNNING threads across projects=%+v", wal.Size(), threads)
}
