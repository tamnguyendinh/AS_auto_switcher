  34: // formatFraction formats the fraction [0.0, 1.0] as a percentage string.
  35: func formatFraction(fraction float64) string {
  36: 	if fraction < 0 {
  37: 		return "--%"
  38: 	}
  39: 	if fraction >= 1.0 {
  40: 		return "100%"
  41: 	}
  42: 	if fraction <= 0.0 {
  43: 		return "0%"
  44: 	}
  45: 	pct := fraction * 100.0
  46: 	if pct == float64(int(pct)) {
  47: 		return fmt.Sprintf("%.0f%%", pct)
  48: 	}
  49: 	return fmt.Sprintf("%.1f%%", pct)
  50: }

  62: // extractFamilyCycles extracts 5h and weekly quota fractions for one model family (Claude or Gemini).
  63: // It respects explicit cycle fractions (>= 0), checks raw buckets, and only falls back to legacy single-bucket fraction
  64: // if no cycle-specific data exists. It never synthesizes a weekly cycle from a legacy single-bucket fraction.
  65: func extractFamilyCycles(bucket telemetry.ModelQuotaBucket, rawBuckets []telemetry.ModelQuotaBucket, familyKeywords []string) (h5 float64, week float64) {
  66: 	h5 = -1
  67: 	week = -1
  68: 
  69: 	// Explicit cycle fractions (>= 0) always take precedence regardless of RemainingFraction.
  70: 	if bucket.FiveHourFraction >= 0 {
  71: 		h5 = bucket.FiveHourFraction
  72: 	}
  73: 	if bucket.WeeklyFraction >= 0 {
  74: 		week = bucket.WeeklyFraction
  75: 	}
  76: 
  77: 	// Supplement missing cycles from rawBuckets
  78: 	for _, b := range rawBuckets {
  79: 		if b.RemainingFraction < 0 && b.FiveHourFraction < 0 && b.WeeklyFraction < 0 {
  80: 			continue
  81: 		}
  82: 		matches := false
  83: 		lower := strings.ToLower(b.ModelID)
  84: 		lowerToken := strings.ToLower(b.TokenType)
  85: 		for _, kw := range familyKeywords {
  86: 			if strings.Contains(lower, kw) || strings.Contains(lowerToken, kw) {
  87: 				matches = true
  88: 				break
  89: 			}
  90: 		}
  91: 		if !matches {
  92: 			continue
  93: 		}
  94: 
  95: 		is5h := strings.Contains(lower, "5h") || b.FiveHourFraction >= 0
  96: 		isWeek := strings.Contains(lower, "week") || b.WeeklyFraction >= 0
  97: 
  98: 		frac := b.RemainingFraction
  99: 		if is5h && h5 < 0 {
 100: 			if b.FiveHourFraction >= 0 {
 101: 				h5 = b.FiveHourFraction
 102: 			} else if frac >= 0 {
 103: 				h5 = frac
 104: 			}
 105: 		}
 106: 		if isWeek && week < 0 {
 107: 			if b.WeeklyFraction >= 0 {
 108: 				week = b.WeeklyFraction
 109: 			} else if frac >= 0 {
 110: 				week = frac
 111: 			}
 112: 		}
 113: 	}
 114: 
 115: 	// Legacy single-bucket fallback: only if NO cycle-specific data found
 116: 	if h5 < 0 && week < 0 && bucket.RemainingFraction >= 0 {
 117: 		h5 = bucket.RemainingFraction
 118: 		week = -1
 119: 	}
 120: 
 121: 	return h5, week
 122: }

 124: // ExtractDualCycleQuota parses 5h and weekly buckets from AccountQuotaData.
 125: func ExtractDualCycleQuota(quota *telemetry.AccountQuotaData) DualCycleQuota {
 126: 	if quota == nil {
 127: 		return DualCycleQuota{Claude5h: -1, ClaudeWeek: -1, Gemini5h: -1, GeminiWeek: -1, HasData: false}
 128: 	}
 129: 	if quota.Status == telemetry.StatusError || quota.ErrorMessage != "" {
 130: 		return DualCycleQuota{Claude5h: -1, ClaudeWeek: -1, Gemini5h: -1, GeminiWeek: -1, HasData: false, IsError: true}
 131: 	}
 132: 
 133: 	c5h, cWeek := extractFamilyCycles(quota.Claude, quota.RawBuckets, []string{"claude", "3p"})
 134: 	g5h, gWeek := extractFamilyCycles(quota.Gemini, quota.RawBuckets, []string{"gemini"})
 135: 
 136: 	dq := DualCycleQuota{
 137: 		Claude5h:   c5h,
 138: 		ClaudeWeek: cWeek,
 139: 		Gemini5h:   g5h,
 140: 		GeminiWeek: gWeek,
 141: 		HasData:    c5h >= 0 || cWeek >= 0 || g5h >= 0 || gWeek >= 0,
 142: 	}
 143: 
 144: 	return dq
 145: }

  35: 		return
  36: 	}
  37: 	titlePtr := windows.StringToUTF16Ptr(title)

  46: 	flags := uintptr(swpNoMove | swpNoSize)
  47: 	_, _, _ = procSetWindowPos.Call(hwnd, insertAfter, 0, 0, 0, 0, flags)

 259: 		return &UIPreferences{Theme: ThemeDark}, nil

 277: 		pref = &UIPreferences{Theme: ThemeDark}

 285: 		return fmt.Errorf("failed creating directory for ui preferences %s: %w", targetDir, err)
 286: 	}
 287: 
 288: 	tmpFile, err := os.CreateTemp(targetDir, ".ui_preferences_tmp_*.json")
 289: 	if err != nil {
 290: 		return fmt.Errorf("failed creating temporary preferences file: %w", err)
 291: 	}
 292: 	tmpPath := tmpFile.Name()
 293: 
 294: 	success := false
 295: 	defer func() {
 296: 		if !success {
 297: 			_ = tmpFile.Close()
 298: 			_ = os.Remove(tmpPath)
 299: 		}
 300: 	}()
 301: 
 302: 	encoder := json.NewEncoder(tmpFile)
 303: 	encoder.SetIndent("", "  ")
 304: 	if err := encoder.Encode(pref); err != nil {
 305: 		return fmt.Errorf("failed encoding ui preferences: %w", err)
 306: 	}
 307: 
 308: 	if err := tmpFile.Sync(); err != nil {
 309: 		return fmt.Errorf("failed flushing temporary preferences file: %w", err)
 310: 	}
 311: 
 312: 	if err := tmpFile.Close(); err != nil {
 313: 		return fmt.Errorf("failed closing temporary preferences file: %w", err)
 314: 	}
 315: 
 316: 	if err := os.Rename(tmpPath, path); err != nil {
 317: 		_ = os.Remove(path)
 318: 		if retryErr := os.Rename(tmpPath, path); retryErr != nil {
 319: 			return fmt.Errorf("failed replacing preferences file %s: %w", path, retryErr)
 320: 		}
 321: 	}
 322: 
 323: 	success = true
 324: 	return nil
 325: }

  88: func readPEFileVersion(exePath string) (string, error) {
  89: 	if runtime.GOOS != "windows" {
  90: 		return "", errors.New("PE version inspection only supported on windows")
  91: 	}
  92: 	cleanPath := filepath.Clean(strings.TrimSpace(exePath))
  93: 	if cleanPath == "" {
  94: 		return "", errors.New("empty executable path")
  95: 	}
  96: 	if _, err := os.Stat(cleanPath); err != nil {
  97: 		return "", err
  98: 	}
  99: 
 100: 	utfPath, err := windows.UTF16PtrFromString(cleanPath)
 101: 	if err != nil {
 102: 		return "", err
 103: 	}
 104: 
 105: 	var dummy uint32
 106: 	size, _, _ := procGetFileVersionInfoSizeW.Call(uintptr(unsafe.Pointer(utfPath)), uintptr(unsafe.Pointer(&dummy)))
 107: 	if size == 0 {
 108: 		return "", errors.New("GetFileVersionInfoSizeW returned 0")
 109: 	}
 110: 
 111: 	buf := make([]byte, size)
 112: 	r1, _, _ := procGetFileVersionInfoW.Call(uintptr(unsafe.Pointer(utfPath)), 0, size, uintptr(unsafe.Pointer(&buf[0])))
 113: 	if r1 == 0 {
 114: 		return "", errors.New("GetFileVersionInfoW failed")
 115: 	}
 116: 
 117: 	subBlock, err := windows.UTF16PtrFromString(`\`)
 118: 	if err != nil {
 119: 		return "", err
 120: 	}
 121: 
 122: 	var ptr uintptr
 123: 	var length uint32
 124: 	r1, _, _ = procVerQueryValueW.Call(
 125: 		uintptr(unsafe.Pointer(&buf[0])),
 126: 		uintptr(unsafe.Pointer(subBlock)),
 127: 		uintptr(unsafe.Pointer(&ptr)),
 128: 		uintptr(unsafe.Pointer(&length)),
 129: 	)

 141: 		return fmt.Sprintf("%d.%d.%d", major, minor, build), nil
 142: 	}
 143: 	return fmt.Sprintf("%d.%d.%d.%d", major, minor, build, rev), nil

 384: func parseResetTime(raw json.RawMessage) time.Time {
 385: 	if len(raw) == 0 {
 386: 		return time.Time{}
 387: 	}
 388: 	trimmed := strings.TrimSpace(string(raw))
 389: 	if trimmed == "" || trimmed == "null" {
 390: 		return time.Time{}
 391: 	}
 392: 
 393: 	// 1. String representation: "2026-10-17T11:29:41Z"
 394: 	if strings.HasPrefix(trimmed, "\"") && strings.HasSuffix(trimmed, "\"") {
 395: 		var s string
 396: 		if err := json.Unmarshal(raw, &s); err == nil {
 397: 			if t, err := time.Parse(time.RFC3339Nano, s); err == nil {
 398: 				return t.UTC()
 399: 			}
 400: 			if t, err := time.Parse(time.RFC3339, s); err == nil {
 401: 				return t.UTC()
 402: 			}
 403: 			if t, err := time.Parse("2006-01-02 15:04:05Z", s); err == nil {
 404: 				return t.UTC()
 405: 			}
 406: 		}
 407: 	}
 408: 
 409: 	// 2. Protobuf JSON object: {"seconds": 1792056078, "nanos": 0}
 410: 	if strings.HasPrefix(trimmed, "{") {
 411: 		var protoTs struct {
 412: 			Seconds int64 `json:"seconds"`
 413: 			Nanos   int64 `json:"nanos"`
 414: 		}
 415: 		if err := json.Unmarshal(raw, &protoTs); err == nil && protoTs.Seconds > 0 {
 416: 			return time.Unix(protoTs.Seconds, protoTs.Nanos).UTC()
 417: 		}
 418: 	}
 419: 
 420: 	// 3. Epoch numeric seconds: 1792056078
 421: 	var epoch int64
 422: 	if err := json.Unmarshal(raw, &epoch); err == nil && epoch > 0 {
 423: 		return time.Unix(epoch, 0).UTC()
 424: 	}
 425: 
 426: 	return time.Time{}
 427: }

 472: // AggregateBuckets parses raw buckets into standard Claude, Gemini, and GPT families, and computes status.
 473: func AggregateBuckets(rawBuckets []ModelQuotaBucket) (claude, gemini, gpt ModelQuotaBucket, canonicalReset time.Time, status string) {
 474: 	claude = ModelQuotaBucket{ModelID: "claude-4-6", RemainingFraction: -1, FiveHourFraction: -1, WeeklyFraction: -1}
 475: 	gemini = ModelQuotaBucket{ModelID: "gemini-2-5", RemainingFraction: -1, FiveHourFraction: -1, WeeklyFraction: -1}
 476: 	gpt = ModelQuotaBucket{ModelID: "gpt-oss", RemainingFraction: -1, FiveHourFraction: -1, WeeklyFraction: -1}
 477: 
 478: 	hasClaude := false
 479: 	hasGemini := false
 480: 	hasGPT := false
 481: 
 482: 	for _, b := range rawBuckets {
 483: 		lower := strings.ToLower(b.ModelID)
 484: 		switch {
 485: 		case strings.Contains(lower, "claude"):
 486: 			if !hasClaude || (b.RemainingFraction >= 0 && (claude.RemainingFraction < 0 || b.RemainingFraction < claude.RemainingFraction)) {
 487: 				claude = b
 488: 				hasClaude = true
 489: 			}
 490: 		case strings.Contains(lower, "gemini"):
 491: 			if !hasGemini || (b.RemainingFraction >= 0 && (gemini.RemainingFraction < 0 || b.RemainingFraction < gemini.RemainingFraction)) {
 492: 				gemini = b
 493: 				hasGemini = true
 494: 			}
 495: 		case strings.Contains(lower, "gpt") || strings.Contains(lower, "oss"):
 496: 			if !hasGPT || (b.RemainingFraction >= 0 && (gpt.RemainingFraction < 0 || b.RemainingFraction < gpt.RemainingFraction)) {
 497: 				gpt = b
 498: 				hasGPT = true
 499: 			}
 500: 		}
 501: 	}
 502: 
 503: 	if !hasClaude && !hasGemini && !hasGPT {
 504: 		return claude, gemini, gpt, canonicalReset, ""
 505: 	}
 506: 
 507: 	minFrac := -1.0
 508: 	if hasClaude && claude.RemainingFraction >= 0 {
 509: 		minFrac = claude.RemainingFraction
 510: 		canonicalReset = claude.ResetTime
 511: 	}
 512: 	if hasGemini && gemini.RemainingFraction >= 0 {
 513: 		if minFrac < 0 || gemini.RemainingFraction < minFrac {
 514: 			minFrac = gemini.RemainingFraction
 515: 			if !gemini.ResetTime.IsZero() {
 516: 				canonicalReset = gemini.ResetTime
 517: 			}
 518: 		} else if canonicalReset.IsZero() && !gemini.ResetTime.IsZero() {
 519: 			canonicalReset = gemini.ResetTime
 520: 		}
 521: 	}
 522: 	if hasGPT && gpt.RemainingFraction >= 0 {
 523: 		if minFrac < 0 || gpt.RemainingFraction < minFrac {
 524: 			minFrac = gpt.RemainingFraction
 525: 			if !gpt.ResetTime.IsZero() {
 526: 				canonicalReset = gpt.ResetTime
 527: 			}
 528: 		} else if canonicalReset.IsZero() && !gpt.ResetTime.IsZero() {
 529: 			canonicalReset = gpt.ResetTime
 530: 		}
 531: 	}
 532: 
 533: 	if canonicalReset.IsZero() {
 534: 		for _, b := range rawBuckets {
 535: 			if !b.ResetTime.IsZero() {
 536: 				canonicalReset = b.ResetTime
 537: 				break
 538: 			}
 539: 		}
 540: 	}
 541: 
 542: 	if minFrac < 0 {
 543: 		status = ""
 544: 	} else if minFrac < 0.15 {
 545: 		if (hasClaude && claude.RemainingFraction < 0.15) && (hasGemini && gemini.RemainingFraction < 0.15) {
 546: 			status = StatusDepleted
 547: 		} else if hasGemini && gemini.RemainingFraction < 0.15 {
 548: 			status = StatusLowGemini
 549: 		} else if hasClaude && claude.RemainingFraction < 0.15 {
 550: 			status = StatusLowClaude
 551: 		} else {
 552: 			status = StatusLowGPT
 553: 		}
 554: 	} else if minFrac <= 0.50 {
 555: 		if hasGemini && gemini.RemainingFraction <= 0.50 && (!hasClaude || claude.RemainingFraction > 0.50) {
 556: 			status = StatusLowGemini
 557: 		} else if hasClaude && claude.RemainingFraction <= 0.50 && (!hasGemini || gemini.RemainingFraction > 0.50) {
 558: 			status = StatusLowClaude
 559: 		} else if hasGPT && gpt.RemainingFraction <= 0.50 {
 560: 			status = StatusLowGPT
 561: 		} else {
 562: 			status = StatusLow
 563: 		}
 564: 	} else if (!hasClaude || claude.RemainingFraction >= 1.0) && (!hasGemini || gemini.RemainingFraction >= 1.0) && (!hasGPT || gpt.RemainingFraction >= 1.0) {
 565: 		status = StatusPristine
 566: 	} else {
 567: 		status = StatusPlentiful
 568: 	}
 569: 
 570: 	return claude, gemini, gpt, canonicalReset, status
 571: }

 573: // RetrieveUserQuota queries Google Cloud Code for the user's quota telemetry.
 574: func (c *QuotaClient) RetrieveUserQuota(ctx context.Context, refreshToken string) (*AccountQuotaData, error) {
 575: 	cleanRefreshToken := strings.TrimSpace(refreshToken)
 576: 	if cleanRefreshToken == "" {
 577: 		return nil, ErrEmptyRefreshToken
 578: 	}
 579: 	if c.refresher == nil {
 580: 		return nil, ErrNilRefresher
 581: 	}
 582: 
 583: 	// 1. Refresh OAuth access token
 584: 	tokenResult, err := c.refresher.RefreshToken(ctx, cleanRefreshToken)
 585: 	if err != nil {
 586: 		return nil, fmt.Errorf("%w: failed to refresh access token: %v", ErrQuotaFetchFailed, err)
 587: 	}
 588: 
 589: 	// 2. Build HTTP request
 590: 	req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.endpoint, bytes.NewBufferString("{}"))
 591: 	if err != nil {
 592: 		return nil, fmt.Errorf("%w: failed to create http request: %v", ErrQuotaFetchFailed, err)
 593: 	}
 594: 	req.Header.Set("Authorization", "Bearer "+tokenResult.AccessToken)
 595: 	req.Header.Set("Content-Type", "application/json")
 596: 	req.Header.Set("Accept", "application/json")
 597: 	req.Header.Set("User-Agent", c.UserAgent())
 598: 	req.Header.Set("X-Goog-Api-Client", DefaultGoogApiClient)
 599: 
 600: 	// 3. Execute HTTP request
 601: 	resp, err := c.httpClient.Do(req)
 602: 	if err != nil {
 603: 		return nil, fmt.Errorf("%w: http request failed: %v", ErrQuotaFetchFailed, err)
 604: 	}
 605: 	defer resp.Body.Close()
 606: 
 607: 	bodyBytes, err := io.ReadAll(resp.Body)
 608: 	if err != nil {
 609: 		return nil, fmt.Errorf("%w: failed to read response body: %v", ErrQuotaFetchFailed, err)
 610: 	}
 611: 
 612: 	if resp.StatusCode != http.StatusOK {
 613: 		return nil, fmt.Errorf("%w: server returned status %d: %s", ErrQuotaFetchFailed, resp.StatusCode, string(bodyBytes))
 614: 	}
 615: 
 616: 	// 4. Try summary payload with groups first
 617: 	claude, gemini, _, rawBuckets, resetTime, status, summaryErr := ParseQuotaSummaryWithBuckets(bodyBytes)
 618: 	if summaryErr == nil {
 619: 		var checkResp rawQuotaSummaryResponse
 620: 		if json.Unmarshal(bodyBytes, &checkResp) == nil && (checkResp.Response.Groups != nil || checkResp.Groups != nil) {
 621: 			return &AccountQuotaData{
 622: 				Claude:      claude,
 623: 				Gemini:      gemini,
 624: 				RawBuckets:  rawBuckets,
 625: 				ResetTime:   resetTime,
 626: 				Status:      status,
 627: 				LastUpdated: time.Now().UTC(),
 628: 			}, nil
 629: 		}
 630: 	}
 631: 
 632: 	// 5. Fallback to legacy rawQuotaResponse with top-level "buckets"
 633: 	var rawResp rawQuotaResponse
 634: 	if err := json.Unmarshal(bodyBytes, &rawResp); err != nil {
 635: 		return nil, fmt.Errorf("%w: failed to parse quota json: %v", ErrQuotaFetchFailed, err)
 636: 	}
 637: 
 638: 	rawBuckets = make([]ModelQuotaBucket, 0, len(rawResp.Buckets))
 639: 	for _, rawB := range rawResp.Buckets {
 640: 		rawBuckets = append(rawBuckets, rawB.toModelQuotaBucket())
 641: 	}
 642: 
 643: 	// 6. Aggregate into model families
 644: 	claude, gemini, _, resetTime, status = AggregateBuckets(rawBuckets)
 645: 
 646: 	return &AccountQuotaData{
 647: 		Claude:      claude,
 648: 		Gemini:      gemini,
 649: 		RawBuckets:  rawBuckets,
 650: 		ResetTime:   resetTime,
 651: 		Status:      status,
 652: 		LastUpdated: time.Now().UTC(),
 653: 	}, nil
 654: }

 126: // SaveToDisk atomically writes the current in-memory quota matrix to disk.
 127: // If customPath is provided, it writes to that path instead of c.filePath.
 128: func (c *QuotaCache) SaveToDisk(customPath ...string) error {
 129: 	targetPath := c.filePath
 130: 	if len(customPath) > 0 && strings.TrimSpace(customPath[0]) != "" {
 131: 		targetPath = customPath[0]
 132: 	}
 133: 
 134: 	// Capture memory snapshot under read lock
 135: 	c.mu.RLock()
 136: 	snapshot := QuotaCacheSnapshot{
 137: 		SavedAt:      time.Now().UTC(),
 138: 		AccountCount: len(c.matrix),
 139: 		Matrix:       make(map[string]*AccountQuotaData, len(c.matrix)),
 140: 	}
 141: 	for k, v := range c.matrix {
 142: 		if v != nil {
 143: 			clone := *v
 144: 			snapshot.Matrix[k] = &clone
 145: 		}
 146: 	}
 147: 	c.mu.RUnlock()
 148: 
 149: 	targetDir := filepath.Dir(targetPath)
 150: 	if err := os.MkdirAll(targetDir, 0755); err != nil {
 151: 		return fmt.Errorf("failed to create cache directory %s: %w", targetDir, err)
 152: 	}
 153: 
 154: 	// Atomic write using temp file + flush + rename
 155: 	tmpFile, err := os.CreateTemp(targetDir, ".quota_cache_tmp_*.json")
 156: 	if err != nil {
 157: 		return fmt.Errorf("failed to create temporary cache file: %w", err)
 158: 	}
 159: 	tmpPath := tmpFile.Name()
 160: 
 161: 	success := false
 162: 	defer func() {
 163: 		if !success {
 164: 			_ = tmpFile.Close()
 165: 			_ = os.Remove(tmpPath)
 166: 		}
 167: 	}()
 168: 
 169: 	encoder := json.NewEncoder(tmpFile)
 170: 	encoder.SetIndent("", "  ")
 171: 	if err := encoder.Encode(snapshot); err != nil {
 172: 		return fmt.Errorf("failed to encode quota snapshot: %w", err)
 173: 	}
 174: 
 175: 	if err := tmpFile.Sync(); err != nil {
 176: 		return fmt.Errorf("failed to flush temp cache file: %w", err)
 177: 	}
 178: 
 179: 	if err := tmpFile.Close(); err != nil {
 180: 		return fmt.Errorf("failed to close temp cache file: %w", err)
 181: 	}
 182: 
 183: 	// Atomic rename to final path (with remove fallback on Windows)
 184: 	if err := os.Rename(tmpPath, targetPath); err != nil {
 185: 		_ = os.Remove(targetPath)
 186: 		if retryErr := os.Rename(tmpPath, targetPath); retryErr != nil {
 187: 			return fmt.Errorf("failed to replace quota cache file %s: %w", targetPath, retryErr)
 188: 		}
 189: 	}
 190: 
 191: 	success = true
 192: 	return nil
 193: }

 185: func (p *TelemetryPool) pollFromLanguageServer(ctx context.Context, email string) (*AccountQuotaData, error) {
 186: 	if p.localQuotaFetcher != nil {
 187: 		return p.localQuotaFetcher(ctx, email)
 188: 	}
 189: 
 190: 	// 1. Discover language server (cached)
 191: 	serverInfo, err := p.getServerInfo()
 192: 	if err != nil || serverInfo == nil {
 193: 		return nil, errors.New("language server not running")
 194: 	}
 195: 
 196: 	// 2. Check active user identity
 197: 	client := connector.NewClient(serverInfo)
 198: 	userStatus, err := client.GetUserStatus(ctx)
 199: 	if err != nil || userStatus == nil || strings.TrimSpace(userStatus.Email) == "" {
 200: 		return nil, errors.New("cannot determine active user from language server")
 201: 	}
 202: 
 203: 	// Only active account is queried via Connect-RPC
 204: 	if !strings.EqualFold(strings.TrimSpace(userStatus.Email), strings.TrimSpace(email)) {
 205: 		return nil, errors.New("account is not active in language server")
 206: 	}
 207: 
 208: 	// 3. Query Connect-RPC endpoint
 209: 	url := fmt.Sprintf("%s/exa.language_server_pb.LanguageServerService/RetrieveUserQuotaSummary", serverInfo.BaseURL())
 210: 	req, err := http.NewRequestWithContext(ctx, http.MethodPost, url, bytes.NewBufferString("{}"))
 211: 	if err != nil {
 212: 		return nil, err
 213: 	}
 214: 	req.Header.Set("Content-Type", "application/json")
 215: 	req.Header.Set("X-Csrf-Token", serverInfo.CSRFToken)
 216: 	req.Header.Set("x-codeium-csrf-token", serverInfo.CSRFToken)
 217: 
 218: 	httpClient := &http.Client{Timeout: 3 * time.Second}
 219: 	resp, err := httpClient.Do(req)
 220: 	if err != nil {
 221: 		return nil, err
 222: 	}
 223: 	defer resp.Body.Close()

 226: 		return nil, fmt.Errorf("language server returned HTTP %d", resp.StatusCode)
 227: 	}

 250: func (p *TelemetryPool) pollAccountWithToken(ctx context.Context, email, refreshToken string) (*AccountQuotaData, error) {
 251: 	cleanEmail := strings.TrimSpace(email)
 252: 	cleanToken := strings.TrimSpace(refreshToken)

 254: 	// Hybrid Telemetry Step 1: For active account running in language_server.exe,
 255: 	// query Connect-RPC /exa.language_server_pb.LanguageServerService/RetrieveUserQuotaSummary
 256: 	if data, err := p.pollFromLanguageServer(ctx, cleanEmail); err == nil && data != nil {
 257: 		data.Email = cleanEmail

 259: 		return data, nil
 260: 	}

 292: // PollAccount refreshes quota telemetry for a single account.
 293: func (p *TelemetryPool) PollAccount(ctx context.Context, email string) (*AccountQuotaData, error) {
 294: 	cleanEmail := strings.TrimSpace(email)
 295: 	if cleanEmail == "" {
 296: 		return nil, errors.New("account email cannot be empty")
 297: 	}
 298: 
 299: 	pollCtx := ctx
 300: 	if _, hasDeadline := ctx.Deadline(); !hasDeadline {
 301: 		var cancel context.CancelFunc
 302: 		pollCtx, cancel = context.WithTimeout(ctx, 5*time.Second)
 303: 		defer cancel()
 304: 	}
 305: 
 306: 	refreshToken, err := p.source.GetRefreshToken(cleanEmail)
 307: 	if err != nil {
 308: 		// Even if store lookup fails, check local language server if this is the active user
 309: 		if data, lsErr := p.pollFromLanguageServer(pollCtx, cleanEmail); lsErr == nil && data != nil {
 310: 			data.Email = cleanEmail
 311: 			p.cache.Update(data)
 312: 			_ = p.cache.SaveToDisk()
 313: 			return data, nil
 314: 		}
 315: 
 316: 		errData := &AccountQuotaData{
 317: 			Email:        cleanEmail,
 318: 			Status:       StatusError,
 319: 			ErrorMessage: err.Error(),
 320: 			LastUpdated:  time.Now().UTC(),
 321: 		}
 322: 		p.cache.Update(errData)
 323: 		_ = p.cache.SaveToDisk()
 324: 		return errData, err
 325: 	}
 326: 
 327: 	data, err := p.pollAccountWithToken(pollCtx, cleanEmail, refreshToken)
 328: 	_ = p.cache.SaveToDisk()
 329: 	return data, err
 330: }

 332: // PollAll concurrently queries quota telemetry for all stored accounts.
 333: // Enforces a 5-second per-batch deadline to guarantee total completion in < 2s for normal operations.
 334: // Returns results, count of successful polls, count of errors, and fatal error.
 335: func (p *TelemetryPool) PollAll(ctx context.Context) ([]*AccountQuotaData, int, int, error) {
 336: 	pollCtx := ctx
 337: 	if _, hasDeadline := ctx.Deadline(); !hasDeadline {
 338: 		var cancel context.CancelFunc
 339: 		pollCtx, cancel = context.WithTimeout(ctx, 5*time.Second)
 340: 		defer cancel()
 341: 	}
 342: 
 343: 	accounts, err := p.source.ListAccounts()
 344: 	if err != nil {
 345: 		return nil, 0, 0, fmt.Errorf("failed to list accounts: %w", err)
 346: 	}
 347: 
 348: 	if len(accounts) == 0 {
 349: 		return []*AccountQuotaData{}, 0, 0, nil
 350: 	}
 351: 
 352: 	results := make([]*AccountQuotaData, len(accounts))
 353: 	var wg sync.WaitGroup
 354: 	var successCount atomic.Int32
 355: 	var errorCount atomic.Int32
 356: 
 357: 	for i, acc := range accounts {
 358: 		wg.Add(1)
 359: 		go func(idx int, info AccountTokenInfo) {
 360: 			defer wg.Done()
 361: 			data, err := p.pollAccountWithToken(pollCtx, info.Email, info.RefreshToken)
 362: 			results[idx] = data
 363: 			if err != nil || (data != nil && data.Status == StatusError) {
 364: 				errorCount.Add(1)
 365: 			} else {
 366: 				successCount.Add(1)
 367: 			}
 368: 		}(i, acc)
 369: 	}
 370: 
 371: 	wg.Wait()
 372: 
 373: 	// Persist snapshot to disk
 374: 	_ = p.cache.SaveToDisk()
 375: 
 376: 	return results, int(successCount.Load()), int(errorCount.Load()), nil
 377: }

 209: func runQuotaCLI(w io.Writer, mgr *manager.AccountManager, ctx context.Context) error {
 210: 	cachePath := telemetry.DefaultQuotaCachePath
 211: 	if mgr != nil && mgr.Store() != nil && mgr.Store().GetStoreDir() != "" {
 212: 		cachePath = filepath.Join(mgr.Store().GetStoreDir(), ".quota_cache.json")
 213: 	}
 214: 
 215: 	cache := telemetry.NewQuotaCache(cachePath)
 216: 	_ = cache.LoadFromDisk()
 217: 
 218: 	// Initialize TelemetryPool backed by store
 219: 	var pool *telemetry.TelemetryPool
 220: 	if mgr != nil && mgr.Store() != nil {
 221: 		refresher := connector.NewGoogleOAuthRefresher(nil)
 222: 		client := telemetry.NewQuotaClient(refresher, nil)
 223: 		p, err := telemetry.NewTelemetryPool(client, cache, mgr.Store())
 224: 		if err == nil {
 225: 			pool = p
 226: 		}
 227: 	}
 228: 
 229: 	// Determine if cache refresh is needed
 230: 	var storedAccounts []string
 231: 	if mgr != nil && mgr.Store() != nil {
 232: 		storedAccounts, _ = mgr.Store().ListAccounts()
 233: 	}
 234: 
 235: 	needsRefresh := false
 236: 	if len(storedAccounts) > 0 {
 237: 		for _, email := range storedAccounts {
 238: 			data, exists := cache.Get(email)
 239: 			if !exists || data == nil || time.Since(data.LastUpdated) > 5*time.Minute {
 240: 				needsRefresh = true
 241: 				break
 242: 			}
 243: 		}
 244: 	} else if len(cache.GetAll()) == 0 {
 245: 		needsRefresh = true
 246: 	}
 247:

 249: 		fmt.Fprintln(w, "[INFO] Fetching latest quota telemetry from Google Cloud Code API...")
 250: 		pollCtx, cancel := context.WithTimeout(ctx, 6*time.Second)
 251: 		_, _, _, _ = pool.PollAll(pollCtx)
 252: 		cancel()
 253: 	}
