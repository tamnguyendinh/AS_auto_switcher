package manager

import (
    "context"
    "encoding/json"
    "net/http"
    "net/http/httptest"
    "net/url"
    "os"
    "path/filepath"
    "reflect"
    "strconv"
    "sync/atomic"
    "testing"
    "time"

    "antigravity-switch/internal/connector"
    "antigravity-switch/internal/model"
    "antigravity-switch/internal/reloader"
    "antigravity-switch/internal/store"
)

func TestR3NilClientMustFailForEveryDiscoveryIdentity(t *testing.T) {
    cases := []struct{ name string; pid, port int }{
        {"ordinary-discovery-info", 9999, 23456},
        {"legacy-ui-discovery-info", 12345, 54321},
    }
    for _, scenario := range cases {
        t.Run(scenario.name, func(t *testing.T) {
            profile := t.TempDir()
            t.Setenv("USERPROFILE", profile)
            accounts, err := store.NewAccountStore(filepath.Join(profile, "accounts"))
            if err != nil { t.Fatal(err) }
            target := "target@r3.invalid"
            if err := accounts.SaveAccount(target, r2Token(target)); err != nil { t.Fatal(err) }
            var killed atomic.Bool
            var genuineStatusAfterKill atomic.Int32
            var genuineSends atomic.Int32
            var postKillFinds atomic.Int32
            var discoveryCalls atomic.Int32
            // The supplied Language Server endpoint always belongs to the old identity.
            genuineServer := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.URL.Path == connector.EndpointGetUserStatus {
                    if killed.Load() { genuineStatusAfterKill.Add(1) }
                    _, _ = w.Write([]byte(`{"email":"old@r3.invalid"}`))
                    return
                }
                if r.URL.Path == connector.EndpointSendUserCascadeMessage { genuineSends.Add(1) }
                w.WriteHeader(http.StatusNotFound)
            }))
            defer genuineServer.Close()
            address, _ := url.Parse(genuineServer.URL)
            actualPort, _ := strconv.Atoi(address.Port())
            genuineClient := connector.NewClientWithHTTP(&connector.ServerInfo{PID: 24680, Port: actualPort, CSRFToken: "synthetic"}, genuineServer.Client())
            rel := reloader.NewSessionReloader()
            rel.StoragePath = filepath.Join(profile, "app_storage.json")
            rel.SnapshotCapturer = func(_, _, _ string) (string, []reloader.ActiveThread, error) {
                threads := []reloader.ActiveThread{{ThreadID: "11111111-1111-4111-8111-111111111111", Project: "C:/SyntheticProject"}}
                data, err := json.Marshal(reloader.ActiveThreadsSnapshot{ActiveThreads: threads})
                if err != nil { return "", nil, err }
                path := filepath.Join(profile, "step1-snapshot.json")
                return path, threads, os.WriteFile(path, data, 0600)
            }
            rel.ProcessKiller = func(context.Context) error { killed.Store(true); return nil }
            rel.ServerDiscoverer = func() (*connector.ServerInfo, error) {
                discoveryCalls.Add(1)
                return &connector.ServerInfo{PID: scenario.pid, Port: scenario.port, CSRFToken: "synthetic"}, nil
            }
            var waits []time.Duration
            rel.Sleeper = func(_ context.Context, d time.Duration) error { waits = append(waits, d); return nil }
            // No ContinuationRunner override: exercise the actual RPC continuation implementation.
            mgr, err := NewAccountManager(accounts, rel)
            if err != nil { t.Fatal(err) }
            credential := r2Token("old@r3.invalid")
            mgr.SetWincredHandlers(func() (*model.TokenData, error) { return credential, nil }, func(token *model.TokenData) error { credential = token; return nil })
            mgr.SetClientFinder(func(context.Context) (*connector.Client, error) {
                if killed.Load() { postKillFinds.Add(1); return nil, nil }
                return genuineClient, nil
            })
            result, err := mgr.SwitchAccount(context.Background(), target)
            if result == nil { t.Fatalf("nil result: %v", err) }
            t.Logf("discovery=(pid=%d,port=%d) finder_after_kill=(nil,nil), finds=%d discovery_calls=%d genuine_status_after_kill=%d genuine_sends=%d waits=%v success=%v error=%v summary=%+v", scenario.pid, scenario.port, postKillFinds.Load(), discoveryCalls.Load(), genuineStatusAfterKill.Load(), genuineSends.Load(), waits, result.Success, err, result.ContinuationSummary)
            if !killed.Load() || postKillFinds.Load() != 1 || !reflect.DeepEqual(waits, []time.Duration{8*time.Second}) || ExtractEmailFromToken(credential) != target {
                t.Fatal("fixture did not reach the requested nil-client gate after credential write and 8-second wait")
            }
            if result.Success || err == nil { t.Error("nil client must fail; producing a local success endpoint cannot verify Language Server RAM") }
            if result.ContinuationSummary != nil { t.Error("actual RPC continuation executed after finder returned nil; no genuine RAM identity was confirmed") }
        })
    }
}
