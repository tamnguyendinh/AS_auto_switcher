func TestSupervisorPurgeUIDeadlineAndPartialSummary(t *testing.T) {
    profile := t.TempDir()
    t.Setenv("USERPROFILE", profile)
    target := "ui@t.invalid"
    accounts, err := store.NewAccountStore(filepath.Join(profile, "accounts"))
    if err != nil { t.Fatal(err) }
    token := func(email string) *model.TokenData {
        return &model.TokenData{AccessToken: "synthetic", IDToken: makeTestJWT(email), Expiry: time.Now().Add(time.Hour), AuthMethod: "consumer"}
    }
    if err := accounts.SaveAccount(target, token(target)); err != nil { t.Fatal(err) }
    var credential atomic.Pointer[model.TokenData]
    credential.Store(token("old@t.invalid"))
    var killed atomic.Bool
    var deadlineSeen atomic.Bool
    var contextSamples atomic.Int32
    var sends atomic.Int32
    var refreshFinderCalls atomic.Int32
    var mu sync.Mutex
    var waits []time.Duration
    var rpc []string
    ids := make([]string, 6)
    threads := make([]reloader.ActiveThread, 6)
    for i := range ids {
        ids[i] = fmt.Sprintf("%08x-1111-4111-8111-111111111111", i+1)
        threads[i] = reloader.ActiveThread{ThreadID: ids[i], Project: fmt.Sprintf("C:/Project%d", i%2), IsMainPane: i == 0}
    }
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        switch r.URL.Path {
        case connector.EndpointGetUserStatus:
            _ = json.NewEncoder(w).Encode(map[string]string{"email": manager.ExtractEmailFromToken(credential.Load())})
        case connector.EndpointCascadeModelConfigData:
            _, _ = w.Write([]byte(`{"defaultOverrideModelConfig":{"modelOrAlias":{"model":"UI_REVIEW_MODEL"}}}`))
        case connector.EndpointSmartFocusConversation, connector.EndpointLoadTrajectory, connector.EndpointSendUserCascadeMessage:
            var body map[string]interface{}
            _ = json.NewDecoder(r.Body).Decode(&body)
            id, _ := body["cascadeId"].(string)
            mu.Lock()
            rpc = append(rpc, filepath.Base(r.URL.Path)+":"+id)
            mu.Unlock()
            if r.URL.Path == connector.EndpointSendUserCascadeMessage {
                sends.Add(1)
                if id == ids[2] { w.WriteHeader(http.StatusInternalServerError); return }
            }
            w.WriteHeader(http.StatusOK)
        default:
            w.WriteHeader(http.StatusNotFound)
        }
    }))
    defer server.Close()
    address, _ := url.Parse(server.URL)
    port, _ := strconv.Atoi(address.Port())
    info := &connector.ServerInfo{PID: 42, Port: port, CSRFToken: "synthetic"}
    rel := reloader.NewSessionReloader()
    rel.StoragePath = filepath.Join(profile, "app_storage.json")
    rel.SnapshotCapturer = func(_, _, _ string) (string, []reloader.ActiveThread, error) {
        data, err := json.Marshal(reloader.ActiveThreadsSnapshot{ActiveThreads: threads})
        if err != nil { return "", nil, err }
        path := filepath.Join(profile, "exact-snapshot.json")
        // Memory deliberately differs: the six threads must be loaded from disk.
        return path, nil, os.WriteFile(path, data, 0600)
    }
    sampleContext := func(ctx context.Context) {
        contextSamples.Add(1)
        if _, limited := ctx.Deadline(); limited { deadlineSeen.Store(true) }
        if ctx.Err() != nil { deadlineSeen.Store(true) }
    }
    rel.ProcessKiller = func(ctx context.Context) error { sampleContext(ctx); killed.Store(true); return nil }
    rel.ServerDiscoverer = func() (*connector.ServerInfo, error) { return info, nil }
    rel.Sleeper = func(ctx context.Context, d time.Duration) error {
        sampleContext(ctx)
        mu.Lock(); waits = append(waits, d); mu.Unlock()
        return nil
    }
    mgr, err := manager.NewAccountManager(accounts, rel)
    if err != nil { t.Fatal(err) }
    mgr.SetChromeDir(t.TempDir())
    mgr.SetWincredHandlers(func() (*model.TokenData, error) { return credential.Load(), nil }, func(value *model.TokenData) error { credential.Store(value); return nil })
    mgr.SetClientFinder(func(ctx context.Context) (*connector.Client, error) {
        if sends.Load() == 6 {
            refreshFinderCalls.Add(1)
            if _, limited := ctx.Deadline(); limited { deadlineSeen.Store(true) }
        }
        return connector.NewClientWithHTTP(info, server.Client()), nil
    })
    fyneApp := test.NewApp()
    app := NewAppWithFyne(fyneApp, mgr)
    app.SetTelemetryPool(nil)
    app.Init()
    defer app.Close()
    waitForIdle(app, 3*time.Second)
    app.SwitchAccount(target)
    waitForIdle(app, 5*time.Second)
    var status string
    fyne.DoAndWait(func() { status = app.statusBarLabel.Text })
    mu.Lock()
    observedWaits := append([]time.Duration(nil), waits...)
    observedRPC := append([]string(nil), rpc...)
    mu.Unlock()
    expectedWaits := []time.Duration{8*time.Second, 3*time.Second, 3*time.Second, 3*time.Second, 3*time.Second, 3*time.Second}
    var expectedRPC []string
    for _, id := range ids {
        for _, call := range []string{"SmartFocusConversation", "LoadTrajectory", "SendUserCascadeMessage"} { expectedRPC = append(expectedRPC, call+":"+id) }
    }
    if deadlineSeen.Load() || contextSamples.Load() != 7 || refreshFinderCalls.Load() == 0 { t.Errorf("UI introduced a deadline or failed to complete refresh: deadline=%v samples=%d refresh=%d", deadlineSeen.Load(), contextSamples.Load(), refreshFinderCalls.Load()) }
    if !killed.Load() || sends.Load() != 6 || !reflect.DeepEqual(observedWaits, expectedWaits) || !reflect.DeepEqual(observedRPC, expectedRPC) { t.Errorf("pipeline mismatch: sends=%d waits=%v rpc=%v", sends.Load(), observedWaits, observedRPC) }
    if !strings.Contains(status, "Đã chuyển sang "+target) || !strings.Contains(status, "• Đã tiếp tục 5/6 phiên") { t.Errorf("incorrect partial continuation status: %q", status) }
    t.Logf("deadline=false at 7 switch context checkpoints and %d post-switch refresh finder calls; requested waits=%v (23s virtual budget); sends=%d; status=%s", refreshFinderCalls.Load(), observedWaits, sends.Load(), status)
    if t.Failed() { return }
    if output := os.Getenv("SUPERVISOR_PURGE_UI_IMAGE"); output != "" {
        var captured image.Image
        fyne.DoAndWait(func() { captured = app.win.Canvas().Capture() })
        file, err := os.Create(output)
        if err != nil { t.Fatal(err) }
        if err := png.Encode(file, captured); err != nil { _ = file.Close(); t.Fatal(err) }
        if err := file.Close(); err != nil { t.Fatal(err) }
        t.Logf("Fyne test canvas captured: %s", output)
    }
}
