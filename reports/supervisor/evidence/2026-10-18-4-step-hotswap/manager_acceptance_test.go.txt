func TestSupervisorSwitchRequiresRAMConfirmation(t *testing.T) {
    profile := t.TempDir()
    t.Setenv("USERPROFILE", profile)
    accStore, err := store.NewAccountStore(filepath.Join(profile, "store"))
    if err != nil {
        t.Fatal(err)
    }
    target := "target@supervisor.invalid"
    if err := accStore.SaveAccount(target, &model.TokenData{AccessToken: "synthetic-target", Expiry: time.Now().Add(time.Hour)}); err != nil {
        t.Fatal(err)
    }
    var killed atomic.Bool
    var statusAfterKill atomic.Int32
    var sendCount atomic.Int32
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        switch r.URL.Path {
        case connector.EndpointGetUserStatus:
            if killed.Load() {
                statusAfterKill.Add(1)
            }
            _, _ = w.Write([]byte(`{"email":"old@supervisor.invalid"}`))
        case connector.EndpointCascadeModelConfigData:
            _, _ = w.Write([]byte(`{"defaultOverrideModelConfig":{"modelOrAlias":{"model":"FIXTURE_MODEL"}}}`))
        case connector.EndpointSendUserCascadeMessage:
            sendCount.Add(1)
            w.WriteHeader(http.StatusOK)
        default:
            w.WriteHeader(http.StatusOK)
        }
    }))
    defer server.Close()
    address, _ := url.Parse(server.URL)
    port, _ := strconv.Atoi(address.Port())
    info := &connector.ServerInfo{PID: 202, Port: port, CSRFToken: "fixture"}
    rel := reloader.NewSessionReloader()
    rel.StoragePath = filepath.Join(profile, "app_storage.json")
    rel.SnapshotCapturer = func(_, _, _ string) (string, []reloader.ActiveThread, error) {
        path := filepath.Join(profile, "snapshot.json")
        threads := []reloader.ActiveThread{{ThreadID: "11111111-1111-4111-8111-111111111111"}}
        data, err := json.Marshal(reloader.ActiveThreadsSnapshot{ActiveThreads: threads})
        if err != nil {
            return "", nil, err
        }
        return path, threads, os.WriteFile(path, data, 0600)
    }
    rel.ProcessKiller = func(context.Context) error { killed.Store(true); return nil }
    rel.ServerDiscoverer = func() (*connector.ServerInfo, error) { return info, nil }
    var waits []time.Duration
    rel.Sleeper = func(_ context.Context, d time.Duration) error { waits = append(waits, d); return nil }
    mgr, err := NewAccountManager(accStore, rel)
    if err != nil {
        t.Fatal(err)
    }
    mgr.SetWincredHandlers(func() (*model.TokenData, error) { return nil, nil }, func(*model.TokenData) error { return nil })
    mgr.SetClientFinder(func(context.Context) (*connector.Client, error) {
        return connector.NewClientWithHTTP(info, server.Client()), nil
    })
    result, err := mgr.SwitchAccount(context.Background(), target)
    t.Logf("RAM stays old; result=%+v error=%v status_after_kill=%d sends=%d waits=%v", result, err, statusAfterKill.Load(), sendCount.Load(), waits)
    if statusAfterKill.Load() == 0 {
        t.Error("new Language Server identity was never checked via GetUserStatus")
    }
    if err == nil && result != nil && result.Success {
        t.Error("SwitchAccount declared success while RAM still belongs to the old account")
    }
    if sendCount.Load() > 0 {
        t.Error("continuation was sent before confirming the new RAM identity")
    }
}

func TestSupervisorDiskSnapshotAndEightSecondOrder(t *testing.T) {
    profile := t.TempDir()
    t.Setenv("USERPROFILE", profile)
    accStore, err := store.NewAccountStore(filepath.Join(profile, "store"))
    if err != nil {
        t.Fatal(err)
    }
    target := "target@supervisor.invalid"
    if err := accStore.SaveAccount(target, &model.TokenData{AccessToken: "synthetic-target", AuthMethod: "consumer", Expiry: time.Now().Add(time.Hour)}); err != nil {
        t.Fatal(err)
    }
    rel := reloader.NewSessionReloader()
    rel.StoragePath = filepath.Join(profile, "app_storage.json")
    var events []string
    var waits []time.Duration
    var continued []reloader.ActiveThread
    diskThreads := []reloader.ActiveThread{
        {ThreadID: "22222222-2222-4222-8222-222222222222", Project: "C:/DiskProject", IsMainPane: true, Model: "disk-model"},
        {ThreadID: "33333333-3333-4333-8333-333333333333", Project: "D:/AnotherProject"},
    }
    rel.SnapshotCapturer = func(_, _, _ string) (string, []reloader.ActiveThread, error) {
        events = append(events, "snapshot")
        path := filepath.Join(profile, "exact-step1-snapshot.json")
        data, err := json.Marshal(reloader.ActiveThreadsSnapshot{ActiveThreads: diskThreads})
        if err != nil {
            return "", nil, err
        }
        err = os.WriteFile(path, data, 0600)
        // Deliberately disagree with the file to prove Step 3 reads disk.
        return path, []reloader.ActiveThread{{ThreadID: "11111111-1111-4111-8111-111111111111"}}, err
    }
    rel.ProcessKiller = func(context.Context) error { events = append(events, "kill"); return nil }
    rel.Sleeper = func(_ context.Context, d time.Duration) error {
        events = append(events, "wait:"+d.String())
        waits = append(waits, d)
        return nil
    }
    rel.ServerDiscoverer = func() (*connector.ServerInfo, error) {
        events = append(events, "discover")
        return &connector.ServerInfo{PID: 303, Port: 42424, CSRFToken: "fixture"}, nil
    }
    rel.ContinuationRunner = func(_ context.Context, _ *connector.Client, threads []reloader.ActiveThread, _ string) error {
        events = append(events, "continue")
        continued = threads
        return nil
    }
    mgr, err := NewAccountManager(accStore, rel)
    if err != nil {
        t.Fatal(err)
    }
    mgr.SetWincredHandlers(func() (*model.TokenData, error) { return nil, nil }, func(*model.TokenData) error {
        events = append(events, "write-credential")
        data, err := os.ReadFile(rel.StoragePath)
        if err != nil {
            return err
        }
        var storage map[string]interface{}
        if err := json.Unmarshal(data, &storage); err != nil {
            return err
        }
        if storage["jetski.onboarding.lastLoginUsername"] != target || storage["jetski.onboarding.lastLoginIsGcpTos"] != "false" {
            t.Errorf("ToS not synchronized before credential write: %+v", storage)
        }
        return nil
    })
    var finderCalls int
    mgr.SetClientFinder(func(context.Context) (*connector.Client, error) {
        finderCalls++
        if finderCalls == 1 {
            return nil, nil
        }
        events = append(events, "new-client")
        return connector.NewClient(&connector.ServerInfo{PID: 303, Port: 42424, CSRFToken: "fixture"}), nil
    })
    result, err := mgr.SwitchAccount(context.Background(), target)
    expected := []string{"snapshot", "write-credential", "kill", "wait:8s", "discover", "new-client", "continue"}
    if err != nil || result == nil || !result.Success || !reflect.DeepEqual(events, expected) || !reflect.DeepEqual(waits, []time.Duration{8*time.Second}) || !reflect.DeepEqual(continued, diskThreads) || !reflect.DeepEqual(result.ActiveThreads, diskThreads) {
        t.Fatalf("snapshot or wait ordering failed: events=%v waits=%v continued=%+v result=%+v err=%v", events, waits, continued, result, err)
    }
    if result.ContinuationSummary == nil || result.ContinuationSummary.SuccessCount != 2 {
        t.Fatalf("continuation summary missing: %+v", result.ContinuationSummary)
    }
    t.Logf("ordered events=%v; continued exact disk snapshot despite differing memory result", events)
}
